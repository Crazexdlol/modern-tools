> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the Messages feature per `docs/superpowers/specs/2026-08-26-messages-design.md`: chat-app-style thread + composer as a new tab on the claim workspace (Claim Info / Documents / **Messages**), a per-user unread queue at `/messages`, and a Messages button on the Dashboard with the caller's unread count.

Reassignment structurally cannot touch messages — no assignment-linked columns.

**Tech Stack:** Same as the rest of the app — Next.js 16 App Router, TypeScript 5, React 19, `@supabase/ssr`, `lucide-react` (MessageSquare icon). No new dependencies.

**Verification model:** No test framework in this repo. "Verify" = `npx tsc --noEmit` + `npm run build` + `npm run lint` (no new warnings) + manual dev-server smoke test where UI is involved. Same standing pattern used for prior features.

  id text primary key,
  claim_id text not null references public.acv_history(id) on delete cascade,
  sender_id uuid references auth.users(id) on delete set null,
  sender_name text not null,
  recipient_ids uuid[] not null,
  recipient_names text[] not null,
  body text not null,
  created_at timestamptz not null default now()
);

create index if not exists messages_recipient_gin on public.messages using gin(recipient_ids);

alter table public.messages enable row level security;

drop policy if exists messages_select on public.messages;
drop policy if exists messages_insert on public.messages;

-- SELECT: caller has access to the parent claim.
create policy messages_select on public.messages for select
  using (
    exists (
      select 1 from public.acv_history c
      where c.id = messages.claim_id
        and (public.is_admin() or c.assigned_to = auth.uid())
    )
  );

-- INSERT: same access + stamps self as sender.
create policy messages_insert on public.messages for insert
  with check (
    sender_id = auth.uid()
    and exists (
      select 1 from public.acv_history c
      where c.id = messages.claim_id
        and (public.is_admin() or c.assigned_to = auth.uid())
    )
  );

-- No UPDATE or DELETE policies — append-only, default-deny.

-- 2. message_reads table (per-user, per-claim high-water-mark)
create table if not exists public.message_reads (
  user_id uuid not null references auth.users(id) on delete cascade,
  claim_id text not null references public.acv_history(id) on delete cascade,
  last_read_at timestamptz not null default now(),
  primary key (user_id, claim_id)
);

alter table public.message_reads enable row level security;

drop policy if exists message_reads_self_all on public.message_reads;

create policy message_reads_self_all on public.message_reads for all
  using (user_id = auth.uid())
  with check (user_id = auth.uid());

-- 3. Verification query (inspect the returned rows manually)
select 'messages table exists' as label, count(*) as value
  from information_schema.tables where table_schema = 'public' and table_name = 'messages'
union all
select 'message_reads table exists', count(*)

union all
select 'messages policies', count(*) from pg_policies where schemaname = 'public' and tablename = 'messages'

Path: `src/lib/auth/require-claim-access.ts`

```typescript
import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'
import { NextResponse } from 'next/server'
import type { Role } from '@/lib/supabase/profiles'

interface CallerContext {
  userId: string
  role: Role
}

// Returns success only if the caller is authenticated AND has access to the
// claim (admin, or the claim's assignee). Otherwise returns a 401/403/404
// response ready to return from the route.
export async function requireClaimAccess(claimId: string): Promise<
  { caller: CallerContext; errorResponse: null } | { caller: null; errorResponse: NextResponse }
> {
  const cookieStore = await cookies()
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() { return cookieStore.getAll() },
        setAll() { /* API routes never set cookies via this path */ },
      },
    }
  )

  const { data: { user } } = await supabase.auth.getUser()
  if (!user) {
    return { caller: null, errorResponse: NextResponse.json({ error: 'Not authenticated' }, { status: 401 }) }
  }

  const { data: profile } = await supabase.from('profiles').select('id, role').eq('id', user.id).maybeSingle()
  if (!profile) {
    return { caller: null, errorResponse: NextResponse.json({ error: 'No profile' }, { status: 403 }) }
  }

  const role = profile.role as Role
  if (role !== 'admin') {
    // Non-admins must be the assignee.
    const { data: claim } = await supabase
      .from('acv_history')
      .select('assigned_to')
      .eq('id', claimId)
      .maybeSingle()
    if (!claim) {
      return { caller: null, errorResponse: NextResponse.json({ error: 'Claim not found' }, { status: 404 }) }
    }
    if (claim.assigned_to !== user.id) {
      return { caller: null, errorResponse: NextResponse.json({ error: 'No access to this claim' }, { status: 403 }) }
    }
  }

  return { caller: { userId: user.id, role }, errorResponse: null }
}

export interface Message {
  id: string
  claimId: string
  senderId: string | null
  senderName: string
  recipientIds: string[]
  recipientNames: string[]
  body: string
  createdAt: string  // ISO
}

export interface UnreadClaim {
  claimId: string
  maNumber: string
  insuredName: string
  vehicle: string
  unreadCount: number
  lastMessageAt: string    // ISO
  lastMessageFrom: string
}

export interface SendMessagePayload {
  recipientIds: string[]
  body: string
}

// Fetch all messages on a claim, oldest-first.
export async function fetchMessages(claimId: string): Promise<Message[]> {
  const res = await fetch(`/api/claims/${encodeURIComponent(claimId)}/messages`)
  if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error ?? 'Failed to load messages')
  const data = await res.json()
  return data.messages as Message[]
}

// Send a new message on a claim. Returns the persisted message row.
export async function sendMessage(claimId: string, payload: SendMessagePayload): Promise<Message> {
  const res = await fetch(`/api/claims/${encodeURIComponent(claimId)}/messages`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  })
  if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error ?? 'Failed to send message')
  const data = await res.json()
  return data.message as Message
}

// Mark this claim's thread as read for the caller (upserts message_reads).
export async function markMessagesRead(claimId: string): Promise<void> {
  await fetch(`/api/claims/${encodeURIComponent(claimId)}/messages/read`, { method: 'POST' })
}

// Fetch the caller's unread queue — list of claims with unread messages.
export async function fetchUnreadQueue(): Promise<UnreadClaim[]> {
  const res = await fetch('/api/messages/queue')
  if (!res.ok) return []
  const data = await res.json()
  return (data.claims ?? []) as UnreadClaim[]
}

// Just the total unread count (sum of unreadCount across claims), for the
// Dashboard header badge. Uses the same queue endpoint.
export async function fetchUnreadCount(): Promise<number> {
  const claims = await fetchUnreadQueue()
  return claims.reduce((sum, c) => sum + c.unreadCount, 0)
}

import { NextRequest, NextResponse } from 'next/server'
import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'
import { requireClaimAccess } from '@/lib/auth/require-claim-access'

export const runtime = 'nodejs'

const MAX_BODY_LEN = 4000

interface MessageRow {
  id: string
  claim_id: string
  sender_id: string | null
  sender_name: string
  recipient_ids: string[]
  recipient_names: string[]
  body: string
  created_at: string
}

function rowToMessage(r: MessageRow) {
  return {
    id: r.id,
    claimId: r.claim_id,
    senderId: r.sender_id,
    senderName: r.sender_name,
    recipientIds: r.recipient_ids,
    recipientNames: r.recipient_names,
    body: r.body,
    createdAt: r.created_at,
  }
}

async function getSupabaseFromCookies() {
  const cookieStore = await cookies()
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    { cookies: { getAll() { return cookieStore.getAll() }, setAll() {} } }
  )
}

// GET /api/claims/[id]/messages — list all messages on the claim, oldest-first.
export async function GET(_req: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params
  const { errorResponse } = await requireClaimAccess(id)
  if (errorResponse) return errorResponse

  const supabase = await getSupabaseFromCookies()
  const { data, error } = await supabase
    .from('messages')
    .select('*')
    .eq('claim_id', id)
    .order('created_at', { ascending: true })
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })

  return NextResponse.json({ messages: (data as MessageRow[]).map(rowToMessage) })
}

// POST /api/claims/[id]/messages — send a message.
// Body: { recipientIds: string[], body: string }
export async function POST(request: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  const { id: claimId } = await params
  const { caller, errorResponse } = await requireClaimAccess(claimId)
  if (errorResponse) return errorResponse

  const body = await request.json().catch(() => null) as { recipientIds?: unknown; body?: unknown } | null
  if (!body) return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 })

  const text = typeof body.body === 'string' ? body.body.trim() : ''
  if (!text) return NextResponse.json({ error: 'Message body is required' }, { status: 400 })
  if (text.length > MAX_BODY_LEN) {
    return NextResponse.json({ error: `Message is too long (max ${MAX_BODY_LEN} characters)` }, { status: 400 })
  }

  const recipientIds = Array.isArray(body.recipientIds)
    ? (body.recipientIds.filter((v): v is string => typeof v === 'string'))
    : []
  if (recipientIds.length === 0) {
    return NextResponse.json({ error: 'At least one recipient is required' }, { status: 400 })
  }

  const supabase = await getSupabaseFromCookies()

  // Look up sender + recipient profiles for the captured display names.
  const allIds = Array.from(new Set([caller!.userId, ...recipientIds]))
  const { data: profs, error: profErr } = await supabase
    .from('profiles')
    .select('id, name, email')
    .in('id', allIds)
  if (profErr) return NextResponse.json({ error: profErr.message }, { status: 400 })

  const profMap = new Map<string, { name: string; email: string }>()
  for (const p of profs ?? []) profMap.set(p.id as string, { name: (p.name as string) || '', email: (p.email as string) || '' })

  const senderProf = profMap.get(caller!.userId)
  if (!senderProf) return NextResponse.json({ error: 'Sender profile not found' }, { status: 403 })

  // Validate every recipient id is a real profile.
  const missing = recipientIds.filter(rid => !profMap.has(rid))
  if (missing.length > 0) {
    return NextResponse.json({ error: `Unknown recipient(s): ${missing.join(', ')}` }, { status: 400 })
  }

  const senderName = senderProf.name || senderProf.email || 'Unknown'
  const recipientNames = recipientIds.map(rid => {
    const p = profMap.get(rid)!
    return p.name || p.email || 'Unknown'
  })

  const id = crypto.randomUUID()
  const { data: inserted, error: insertErr } = await supabase
    .from('messages')
    .insert({
      id,
      claim_id: claimId,
      sender_id: caller!.userId,
      sender_name: senderName,
      recipient_ids: recipientIds,
      recipient_names: recipientNames,
      body: text,
    })
    .select('*')
    .maybeSingle()
  if (insertErr || !inserted) {
    return NextResponse.json({ error: insertErr?.message ?? 'Insert failed' }, { status: 400 })
  }

  return NextResponse.json({ message: rowToMessage(inserted as MessageRow) })
}

}

import { NextRequest, NextResponse } from 'next/server'
import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'

export const runtime = 'nodejs'

interface MessageQueueRow {
  id: string
  claim_id: string
  sender_name: string
  created_at: string
}

interface ClaimRow {
  id: string
  ma_number: string | null
  vehicle: string | null
  claim: { insuredName?: string } | null
}

interface ReadRow {
  claim_id: string
  last_read_at: string
}

// GET /api/messages/queue — list of claims with unread messages for the caller.
// Response: { claims: [{ claimId, maNumber, insuredName, vehicle, unreadCount, lastMessageAt, lastMessageFrom }] }
export async function GET(_req: NextRequest) {
  const cookieStore = await cookies()
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    { cookies: { getAll() { return cookieStore.getAll() }, setAll() {} } }
  )

  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 })

  // Fetch every message where the caller is a recipient (RLS still narrows
  // to claims the caller can see — belt + suspenders).
  const { data: msgs, error: msgErr } = await supabase
    .from('messages')
    .select('id, claim_id, sender_name, created_at')
    .contains('recipient_ids', [user.id])
    .order('created_at', { ascending: false })
  if (msgErr) return NextResponse.json({ error: msgErr.message }, { status: 400 })
  const messages = (msgs as MessageQueueRow[]) ?? []

  if (messages.length === 0) return NextResponse.json({ claims: [] })

  // Fetch caller's read timestamps for the relevant claims.
  const claimIds = Array.from(new Set(messages.map(m => m.claim_id)))
  const { data: reads, error: readErr } = await supabase
    .from('message_reads')
    .select('claim_id, last_read_at')
    .eq('user_id', user.id)
    .in('claim_id', claimIds)
  if (readErr) return NextResponse.json({ error: readErr.message }, { status: 400 })
  const readAt = new Map<string, string>()
  for (const r of (reads as ReadRow[]) ?? []) readAt.set(r.claim_id, r.last_read_at)

  // Bucket unread messages per claim.
  const buckets = new Map<string, { count: number; lastAt: string; lastFrom: string }>()
  for (const m of messages) {
    const cutoff = readAt.get(m.claim_id)
    if (cutoff && m.created_at <= cutoff) continue  // already read
    const b = buckets.get(m.claim_id)
    if (!b) {
      buckets.set(m.claim_id, { count: 1, lastAt: m.created_at, lastFrom: m.sender_name })
    } else {
      b.count++
      if (m.created_at > b.lastAt) { b.lastAt = m.created_at; b.lastFrom = m.sender_name }
    }
  }

  if (buckets.size === 0) return NextResponse.json({ claims: [] })

  // Enrich each claim with display fields.
  const unreadClaimIds = Array.from(buckets.keys())
  const { data: claims, error: claimErr } = await supabase
    .from('acv_history')
    .select('id, ma_number, vehicle, claim')
    .in('id', unreadClaimIds)
  if (claimErr) return NextResponse.json({ error: claimErr.message }, { status: 400 })

  const claimMap = new Map<string, ClaimRow>()
  for (const c of (claims as ClaimRow[]) ?? []) claimMap.set(c.id, c)

  const payload = unreadClaimIds
    .map(cid => {
      const b = buckets.get(cid)!
      const c = claimMap.get(cid)
      if (!c) return null  // RLS filtered it out (shouldn't happen if messages_select passed, but safe)
      return {
        claimId: cid,
        maNumber: c.ma_number ?? '',
        insuredName: c.claim?.insuredName ?? '',
        vehicle: c.vehicle ?? '',
        unreadCount: b.count,
        lastMessageAt: b.lastAt,
        lastMessageFrom: b.lastFrom,
      }
    })
    .filter((v): v is NonNullable<typeof v> => v !== null)
    .sort((a, b) => b.lastMessageAt.localeCompare(a.lastMessageAt))

  return NextResponse.json({ claims: payload })
}

    } catch (err) {
      setLoadError(err instanceof Error ? err.message : "Failed to load messages")
    } finally {
      setLoading(false)
    }
  }, [claimId, recipientIds.length])

  // Initial load + fetch profiles + mark-as-read on mount.
  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect -- initial-load pattern; setState fires only after the async fetches resolve
    load()
    // eslint-disable-next-line react-hooks/set-state-in-effect -- initial-load pattern
    fetchAllProfilesBrowser().then(setProfiles)
    markMessagesRead(claimId).catch(() => { /* silent — read state isn't user-visible */ })
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [claimId])

  // Auto-scroll to bottom when new messages arrive.
  useEffect(() => {
    const el = threadRef.current
    if (!el) return
    el.scrollTop = el.scrollHeight
  }, [messages.length])

  // Close picker on click-outside.
  useEffect(() => {
    if (!pickerOpen) return
    const onDoc = (e: MouseEvent) => {
      if (pickerRef.current && !pickerRef.current.contains(e.target as Node)) setPickerOpen(false)
    }
    document.addEventListener("mousedown", onDoc)
    return () => document.removeEventListener("mousedown", onDoc)
  }, [pickerOpen])

  const pickerOptions = useMemo(() => {
    const q = pickerQuery.trim().toLowerCase()
    return profiles
      .filter(p => !recipientIds.includes(p.id))
      .filter(p => !q || (p.name || "").toLowerCase().includes(q) || p.email.toLowerCase().includes(q))
      .slice(0, 12)
  }, [profiles, recipientIds, pickerQuery])

  const selectedProfiles = useMemo(
    () => recipientIds.map(id => profiles.find(p => p.id === id)).filter((p): p is Profile => !!p),
    [recipientIds, profiles],
  )

  const addRecipient = (id: string) => {
    setRecipientIds(prev => (prev.includes(id) ? prev : [...prev, id]))
    setPickerQuery("")
    setPickerOpen(false)
  }
  const removeRecipient = (id: string) => setRecipientIds(prev => prev.filter(x => x !== id))

  const canSend = body.trim().length > 0 && recipientIds.length > 0 && !sending

  const doSend = async () => {
    if (!canSend) return
    setSending(true)
    setSendError(null)
    try {
      const created = await sendMessage(claimId, { recipientIds, body: body.trim() })
      setMessages(prev => [...prev, created])
      setBody("")
      // Keep recipientIds populated — user is likely to send a follow-up.
    } catch (err) {
      setSendError(err instanceof Error ? err.message : "Failed to send")
    } finally {
      setSending(false)
    }
  }

  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    // Cmd/Ctrl + Enter sends
    if ((e.metaKey || e.ctrlKey) && e.key === "Enter") {
      e.preventDefault()
      doSend()
    }
  }

  return (
    <div className="flex flex-col h-[calc(100vh-320px)] min-h-[500px] bg-white border border-gray-200 rounded-lg overflow-hidden">
      {/* Thread header */}
      <div className="flex items-center justify-between px-4 py-2 border-b border-gray-200 bg-gray-50">
        <div className="text-sm text-gray-600">
          {loading ? "Loading…" : `${messages.length} message${messages.length === 1 ? "" : "s"}`}
        </div>
        <button
          onClick={load}
          disabled={loading}
          className="text-gray-400 hover:text-gray-700 p-1 disabled:opacity-40"
          title="Refresh"
        >
          <RefreshCw className={cn("w-4 h-4", loading && "animate-spin")} />
        </button>
      </div>

      {/* Thread */}
      <div ref={threadRef} className="flex-1 overflow-y-auto px-4 py-3 space-y-3">
        {loadError && (
          <div className="p-3 rounded bg-red-50 border border-red-200 text-red-700 text-sm">{loadError}</div>
        )}
        {!loading && messages.length === 0 && !loadError && (
          <div className="text-sm text-gray-400 italic text-center py-8">
            No messages on this claim yet — start the thread below.
          </div>
        )}
        {messages.map(m => (
          <div key={m.id} className="border border-gray-200 rounded-lg bg-white px-3 py-2">
            <div className="text-xs text-gray-500 mb-1">
              <span className="font-semibold text-gray-800">{m.senderName}</span>
              <span className="mx-1">→</span>
              <span>{m.recipientNames.join(", ")}</span>
              <span className="mx-1">·</span>
              <span>{fmtStamp(m.createdAt)}</span>
            </div>
            <div className="text-sm text-gray-800">
              <LinkifiedBody text={m.body} />
            </div>
          </div>
        ))}
      </div>

      {/* Composer */}
      <div className="border-t border-gray-200 p-3 space-y-2 bg-gray-50">
        {/* Recipient chips + picker */}
        <div className="flex flex-wrap items-center gap-2">
          <span className="text-xs text-gray-500 font-semibold">To:</span>
          {selectedProfiles.map(p => (
            <span
              key={p.id}
              className="inline-flex items-center gap-1 bg-blue-100 text-blue-800 text-xs font-medium px-2 py-0.5 rounded"
            >
              {p.name || p.email}
              <button
                type="button"
                onClick={() => removeRecipient(p.id)}
                className="text-blue-500 hover:text-blue-800"
                aria-label={`Remove ${p.name || p.email}`}
              >
                <X className="w-3 h-3" />
              </button>
            </span>
          ))}
          <div className="relative flex-1 min-w-[120px]" ref={pickerRef}>
            <input
              value={pickerQuery}
              onFocus={() => setPickerOpen(true)}
              onChange={(e) => { setPickerQuery(e.target.value); setPickerOpen(true) }}
              placeholder={recipientIds.length === 0 ? "Add recipient…" : "Add another…"}
              className="w-full bg-white border border-gray-200 rounded px-2 py-1 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
            {pickerOpen && pickerOptions.length > 0 && (
              <div className="absolute left-0 top-full mt-1 w-full max-h-56 overflow-y-auto bg-white border border-gray-200 rounded-lg shadow-lg z-30">
                {pickerOptions.map(p => (
                  <button
                    key={p.id}
                    type="button"
                    onClick={() => addRecipient(p.id)}
                    className="w-full text-left px-3 py-2 text-sm text-gray-700 hover:bg-gray-100"
                  >
                    <div className="font-semibold">{p.name || p.email}</div>
                    {p.name && <div className="text-xs text-gray-500">{p.email}</div>}
                  </button>
                ))}
              </div>
            )}
            {pickerOpen && pickerOptions.length === 0 && pickerQuery && (
              <div className="absolute left-0 top-full mt-1 w-full bg-white border border-gray-200 rounded-lg shadow-lg z-30 px-3 py-2 text-sm text-gray-400 italic">
                No users found
              </div>
            )}
          </div>
        </div>

        {/* Body + send */}
        <div className="flex items-end gap-2">
          <textarea
            value={body}
            onChange={(e) => setBody(e.target.value)}
            onKeyDown={onKeyDown}
            placeholder="Write a message… (Ctrl+Enter to send)"
            rows={3}
            className="flex-1 bg-white border border-gray-200 rounded px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500 resize-y max-h-40 min-h-[70px]"
          />
          <Button
            onClick={doSend}
            disabled={!canSend}
            className="bg-blue-600 hover:bg-blue-500 text-white gap-2"
          >
            {sending ? <Loader2 className="w-4 h-4 animate-spin" /> : <Send className="w-4 h-4" />}
            {sending ? "Sending…" : "Send"}
          </Button>
        </div>

        {sendError && (
          <div className="p-2 rounded bg-red-50 border border-red-200 text-red-700 text-xs">{sendError}</div>
        )}
        {recipientIds.length > 20 && (
          <div className="text-xs text-amber-600">Sending to {recipientIds.length} people — double-check the list.</div>
        )}
      </div>
    </div>
  )
}
```

"use client"

import Link from 'next/link'
import { useEffect, useState } from 'react'
import { Loader2 } from 'lucide-react'
import { fetchUnreadQueue, type UnreadClaim } from '@/lib/supabase/messages'

function fmtStamp(iso: string): string {
  return new Date(iso).toLocaleString('en-US', {
    month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit',
  })
}

export function MessagesQueueClient() {
  const [claims, setClaims] = useState<UnreadClaim[] | null>(null)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect -- mount-load pattern; setState fires only after the async fetch resolves
    fetchUnreadQueue()
      .then(setClaims)
      .catch(e => setError(e instanceof Error ? e.message : 'Failed to load queue'))
  }, [])

  if (error) {
    return (
      <div className="p-3 rounded bg-red-50 border border-red-200 text-red-700 text-sm">{error}</div>
    )
  }

  if (claims === null) {
    return (
      <div className="flex items-center gap-2 text-gray-500 text-sm">
        <Loader2 className="w-4 h-4 animate-spin" /> Loading…
      </div>
    )
  }

  if (claims.length === 0) {
    return <div className="text-sm text-gray-500 italic">No unread messages.</div>
  }

  return (
    <div className="bg-white border border-gray-200 rounded-lg shadow-sm overflow-hidden">
      <table className="w-full text-sm">
        <thead>
          <tr className="border-b border-gray-200 bg-gray-50">
            <th className="text-left px-4 py-3 text-xs font-semibold text-gray-500 uppercase tracking-wider">Claim</th>
            <th className="text-left px-4 py-3 text-xs font-semibold text-gray-500 uppercase tracking-wider">Vehicle</th>
            <th className="text-left px-4 py-3 text-xs font-semibold text-gray-500 uppercase tracking-wider">Unread</th>
            <th className="text-left px-4 py-3 text-xs font-semibold text-gray-500 uppercase tracking-wider">Last message</th>
          </tr>
        </thead>
        <tbody>
          {claims.map(c => (
            <tr key={c.claimId} className="border-b border-gray-100 hover:bg-gray-50 transition-colors">
              <td className="px-4 py-3">
                <Link
                  href={`/claims/${encodeURIComponent(c.claimId)}?tab=messages`}
                  className="text-gray-900 hover:text-blue-600 font-medium block"
                >
                  {c.maNumber || c.claimId}
                  {c.insuredName && <span className="text-gray-500 font-normal"> · {c.insuredName}</span>}
                </Link>
              </td>
              <td className="px-4 py-3 text-gray-700">{c.vehicle || '—'}</td>

                  {c.unreadCount}
                </span>
              </td>

                {fmtStamp(c.lastMessageAt)} <span className="text-gray-400">·</span> {c.lastMessageFrom}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}
